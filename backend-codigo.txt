@RestController
@RequestMapping("/api/auth")
@CrossOrigin(origins = "*")
@Validated
public class AuthController {

    @Autowired
    private UsuarioRepository usuarioRepository;

    // Si tenemos una sesiÃ³n en memoria mientras dure la instancia,
    // entonces guardamos acÃ¡ el correo del usuario que hizo login.
    // Es temporal: cuando el front pide /es-admin volvemos a buscar al usuario
    // en la BD para leer su rol de verdad.
    private static String sesionEmail = null;

    private final BCryptPasswordEncoder passwordEncoder = new BCryptPasswordEncoder();

    @PostMapping("/login")
    public ResponseEntity<?> login(@Valid @RequestBody LoginRequest credenciales) {
        String email = credenciales.getEmail().trim().toLowerCase();
        String password = credenciales.getPassword();

        // Si el correo no tiene formato vÃ¡lido, entonces ni buscamos en la BD:
        // devolvemos 400 directo para que el front muestre el error en el formulario.
        if (!ValidacionesUtil.esEmailValido(email)) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                    .body(Map.of("error", "El formato del email es invÃ¡lido"));
        }

        // Si no encontramos el correo, entonces mandamos 401 "credenciales invÃ¡lidas"
        // (no decimos que el usuario no existe, por seguridad).
        Usuario usuario = usuarioRepository.findByEmail(email);
        if (usuario == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(Map.of("error", "Credenciales invÃ¡lidas"));
        }

        // Si la contraseÃ±a que viene no coincide con el hash guardado,
        // entonces tampoco dejamos entrar: usamos BCrypt.matches, nunca texto plano.
        if (!passwordEncoder.matches(password, usuario.getPassword())) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(Map.of("error", "Credenciales invÃ¡lidas"));
        }

        // Si pasamos las dos validaciones, entonces marcamos la sesiÃ³n con este correo.
        sesionEmail = email;

        Map<String, Object> respuesta = new HashMap<>();
        respuesta.put("token", "token-simulado-" + email);
        respuesta.put("usuario", usuario);

        EntityModel<Map<String, Object>> model = EntityModel.of(respuesta);
        model.add(WebMvcLinkBuilder.linkTo(WebMvcLinkBuilder.methodOn(AuthController.class).login(null)).withSelfRel());
        model.add(WebMvcLinkBuilder.linkTo(WebMvcLinkBuilder.methodOn(AuthController.class).sesion()).withRel("sesion"));

        return ResponseEntity.ok(model);
    }

    @PostMapping("/registro")
    public ResponseEntity<?> registro(@Valid @RequestBody RegistroRequest datos) {
        String nombre = datos.getNombre().trim();
        String email = datos.getEmail().trim().toLowerCase();
        String password = datos.getPassword();
        String fechaNacimientoStr = datos.getFechaNacimiento();

        // Si el email no es vÃ¡lido, entonces ni seguimos con el registro.
        if (!ValidacionesUtil.esEmailValido(email)) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                    .body(Map.of("error", "El formato del email es invÃ¡lido"));
        }

        // Si el correo ya estÃ¡ en la BD, entonces no creamos otro usuario:
        // el email es Ãºnico y se lo avisamos al front con un 400.
        if (usuarioRepository.findByEmail(email) != null) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                    .body(Map.of("error", "El email ya estÃ¡ registrado"));
        }

        // Si la fecha viene en otro formato, entonces no la podemos parsear:
        // le pedimos YYYY-MM-DD para que el front sepa cÃ³mo enviarla.
        LocalDate fechaNacimiento;
        try {
            fechaNacimiento = LocalDate.parse(fechaNacimientoStr);
        } catch (Exception e) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                    .body(Map.of("error", "Formato de fecha invÃ¡lido. Use YYYY-MM-DD"));
        }

        // Si la edad no estÃ¡ entre 18 y 120, entonces validarEdad lanza:
        // la capturamos y le mandamos el mensaje al front con un 400.
        try {
            ValidacionesUtil.validarEdad(fechaNacimiento);
        } catch (IllegalArgumentException e) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                    .body(Map.of("error", e.getMessage()));
        }

        // Si llegamos hasta acÃ¡, entonces armamos el usuario:
        // guardamos la contraseÃ±a hasheada con BCrypt y el rol siempre "cliente"
        // (los admin los creamos directo en Neon, no por este endpoint).
        Usuario nuevo = new Usuario();
        nuevo.setNombre(nombre);
        nuevo.setEmail(email);
        nuevo.setPassword(passwordEncoder.encode(password));
        nuevo.setRol("cliente");
        nuevo.setFechaNacimiento(fechaNacimiento);

        usuarioRepository.save(nuevo);

        Map<String, Object> respuesta = new HashMap<>();
        respuesta.put("ok", true);
        respuesta.put("mensaje", "Usuario registrado exitosamente");

        EntityModel<Map<String, Object>> model = EntityModel.of(respuesta);
        model.add(WebMvcLinkBuilder.linkTo(WebMvcLinkBuilder.methodOn(AuthController.class).registro(null)).withSelfRel());
        model.add(WebMvcLinkBuilder.linkTo(WebMvcLinkBuilder.methodOn(AuthController.class).login(null)).withRel("login"));

        return ResponseEntity.status(HttpStatus.CREATED).body(model);
    }

    @GetMapping("/sesion")
    public ResponseEntity<?> sesion() {
        if (sesionEmail == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(Map.of("error", "No hay sesiÃ±n activa"));
        }

        Usuario usuario = usuarioRepository.findByEmail(sesionEmail);
        if (usuario == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(Map.of("error", "SesiÃ±n invÃ±lida"));
        }

        EntityModel<Usuario> model = EntityModel.of(usuario);
        model.add(WebMvcLinkBuilder.linkTo(WebMvcLinkBuilder.methodOn(AuthController.class).sesion()).withSelfRel());
        model.add(WebMvcLinkBuilder.linkTo(WebMvcLinkBuilder.methodOn(AuthController.class).esAdmin()).withRel("es-admin"));

        return ResponseEntity.ok(model);
    }

    @PostMapping("/logout")
    public ResponseEntity<?> logout() {
        sesionEmail = null;
        return ResponseEntity.ok(Map.of("mensaje", "SesiÃ±n cerrada"));
    }

    @GetMapping("/es-admin")
    public ResponseEntity<?> esAdmin() {
        if (sesionEmail == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(Map.of("esAdmin", false));
        }

        Usuario usuario = usuarioRepository.findByEmail(sesionEmail);
        if (usuario == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(Map.of("esAdmin", false));
        }

        // Si hay sesiÃ³n, entonces comparamos el rol del usuario:
        // con esto el front decide si muestra o no el botÃ³n de AdministraciÃ³n.
        boolean esAdmin = "admin".equals(usuario.getRol());
        return ResponseEntity.ok(Map.of("esAdmin", esAdmin));
    }

     * campo -> mensaje para que el front sepa exactamente quÃ© corregir.

        Map<String, String> errores = new HashMap<>();

            errores.put(error.getField(), error.getDefaultMessage())
        );
        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                .body(Map.of("error", "Datos invÃ±lidos", "campos", errores));
    }

 */
public class ValidacionesUtil {

    private static final Pattern EMAIL_PATTERN = Pattern.compile(
        "^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$"
    );

    private ValidacionesUtil() {
        // Clase de utilidad, no se instancia
    }

    /**
     * Valida que el email tenga un formato vÃ¡lido.
     */
    public static boolean esEmailValido(String email) {
        return email != null && EMAIL_PATTERN.matcher(email).matches();
    }

    /**
     * Valida que la edad estÃ© entre 18 y 120 aÃ±os.
     * Lanza IllegalArgumentException si la edad no es vÃ¡lida.
     */
    public static void validarEdad(LocalDate fechaNacimiento) {
        if (fechaNacimiento == null) {
            throw new IllegalArgumentException("La fecha de nacimiento es obligatoria");
        }

        // Usamos Period (no solo restar aÃ±os) para respetar dÃ­a y mes:
        // si todavÃ­a no cumpliÃ³ este aÃ±o, la edad no cuenta completo.
        LocalDate hoy = LocalDate.now();
        int edad = Period.between(fechaNacimiento, hoy).getYears();

        // Si es menor de 18, entonces no puede registrarse (venta +18).
        if (edad < 18) {
            throw new IllegalArgumentException("Debes ser mayor de 18 aÃ±os para registrarte");
        }
        // Si pasa de 120, entonces la fecha es invÃ¡lida (probablemente un typo).
        if (edad > 120) {
            throw new IllegalArgumentException("La edad no puede ser mayor a 120 aÃ±os");
        }
    }
